## 2. 新增 `src/cn_clean.py`

```python
# -*- coding: utf-8 -*-
"""中文 PDF 文本清洗 —— 对应三个实测到的排版瑕疵。

为什么单独一个文件：
    这三处清洗要在三个地方用（p0_1 的表格、p0_2 的正文、chunk_loader 的兜底），
    分散写三遍迟早会不一致。

三个瑕疵（都实测过，见方案文档 §0.2）：
    1. 竖排表头被抽成单字并用 <br> 连：'定<br>时<br>器' -> '定时器'
    2. 汉字之间被排版插入空格：      '定 时 器' -> '定时器'
    3. 零宽字符 / 全角空格 / 连续空格
"""
from __future__ import annotations

import re

CJK = r"\u4e00-\u9fff"

_BR = re.compile(r"<br\s*/?>", re.IGNORECASE)
_ZERO = re.compile(r"[\u200b-\u200f\u202a-\u202e\ufeff]")
_MULTI = re.compile(r"[ \t]{2,}")
# 连续出现"汉字 + 空格 + 汉字"的片段
_CJK_SPACE_RUN = re.compile(rf"[{CJK}](?:[ \t]+[{CJK}])+")


def normalize_br(text: str) -> str:
    """把 PyMuPDF 留下的 <br> 按语义还原。

    判据（实测有效）：
        所有片段都只有 1 个字  -> 它是竖排文字，直接拼起来：'定<br>时<br>器' -> '定时器'
        否则                  -> 它是真换行，用空格分隔：'产品型号<br>资源差异' -> '产品型号 资源差异'
    """
    if not text or "<br" not in text.lower():
        return text or ""
    parts = [p.strip() for p in _BR.split(text)]
    parts = [p for p in parts if p]
    if not parts:
        return ""
    if all(len(p) == 1 for p in parts):
        return "".join(parts)
    return " ".join(parts)


def squeeze_cjk_spaces(text: str, min_frags: int = 3) -> str:
    """去掉"汉字之间被排版插进来的空格"：'定 时 器' -> '定时器'。

    为什么要求 min_frags >= 3（而不是见到"汉字 空格 汉字"就删）：
        如果只要求 2 个片段，'上海 深圳' 会被误拼成 '上海深圳'；
        而竖排表头一定是 3 个以上单字被拆开（'定 时 器'、'系 统 时 间'），
        所以用"连续片段数 >= 3"当判据，既不误伤又不漏。
        —— 实测 '产品型号 资源差异'（2 个片段）和 '定时器 通用'（2 个片段）都不会被误拼。
    """
    def _rep(m: re.Match) -> str:
        seg = m.group(0)
        frags = [f for f in re.split(r"[ \t]+", seg) if f]
        if len(frags) >= min_frags:
            return "".join(frags)
        return seg
    return _CJK_SPACE_RUN.sub(_rep, text)


def clean_cell(text: str) -> str:
    """单元格 / 短文本的清洗。"""
    t = _ZERO.sub("", text or "")
    t = t.replace("\u3000", " ").replace("\xa0", " ")
    t = normalize_br(t)
    t = squeeze_cjk_spaces(t)
    t = _MULTI.sub(" ", t)
    return t.strip()


def has_cjk(text: str) -> bool:
    return bool(re.search(rf"[{CJK}]", text or ""))


def cjk_ratio(text: str) -> float:
    if not text:
        return 0.0
    return len(re.findall(rf"[{CJK}]", text)) / len(text)


def estimate_tokens(text: str) -> int:
    """粗估 token：中文约 1 字 1 token，英文约 4 字符 1 token。

    原代码用的 total_chars / 3.5 是英文经验值 ——
    中文会被低估三四倍（500 个汉字 ≈ 500 token，而不是 143）。
    """
    cjk = len(re.findall(rf"[{CJK}]", text or ""))
    rest = len(text or "") - cjk
    return cjk + max(1, rest // 4)
```

## 3. 新增 `src/table_struct.py`

> 这是整套方案里最值钱的一块。核心思想：**不信 `find_tables()` 给的网格，用单元格坐标自己重建。**

```python
# -*- coding: utf-8 -*-
"""表格结构化：用坐标重建规整网格，不信 find_tables() 给的列。

为什么必须做（实测：CH32F103 第 2 页 表1-1）：
    PyMuPDF 把这 5 列的表认成了 14 列 —— 双线边框和斜线表头都被当成列分隔线，
    每个真实列右边多挂 1~2 条 5.4pt 宽的"碎片列"。to_markdown() 于是输出
        |芯片引脚数|芯片引脚数|48|48|48|48|48|48|48|48|48|64|64|64|
    "48" 重复 12 次，看不出哪个 48 属于哪个型号。

解法（你说的"表格要注意数据按坐标确定"）：
    1. 列边界 = 所有单元格 x0/x1 的并集，间距 < 12pt 的聚成一簇取中位数（消掉碎片列）
    2. 行边界 = 所有单元格 y0/y1 的并集，间隔 < 6pt 的同样处理
    3. 每格文字 = 落在该格里的所有 word（先按"竖排单字链"合并）
    4. 纵向合并：上方文字延伸到本行 -> 继承（'定时器' 跨 4 行）
    5. 横向合并：整行只有一个值、且它跨过了列边界 -> 该值对每列都成立
"""
from __future__ import annotations

import re
from statistics import median

from cn_clean import clean_cell

GC = 12.0              # 竖线聚类间距：小于它的相邻边界合成一簇（碎片列）
GR = 6.0               # 横线聚类间距
MAX_HEADER_ROWS = 3
MAX_VCHAIN = 8         # 竖排单字链最长几字
MAX_LABEL_COLS = 3
CJK1 = re.compile(r"^[\u4e00-\u9fff]$")
VALUE_RE = re.compile(r"^\d[\d.,]*\s*[A-Za-z%/°μΩ]*$")
LABEL_JOIN = " / "


# ---------- 一、坐标重建网格 ----------

def _merge_close(bounds, gap):
    """把间距 < gap 的相邻边界聚成一簇，取中位数作为唯一边界。

    表1-1 的 [261.2, 266.7, 272.0] 三簇间距只有 5.5/5.3 -> 合并成 266.7，
    正是那条真实列边界。
    """
    if not bounds:
        return []
    bs = sorted(bounds)
    out, cur = [], [bs[0]]
    for b in bs[1:]:
        if b - cur[-1] < gap:
            cur.append(b)
        else:
            out.append(median(cur))
            cur = [b]
    out.append(median(cur))
    return out


def _overlap_h(a, b):
    ov = min(a[2], b[2]) - max(a[0], b[0])
    return ov / max(1e-6, min(a[2] - a[0], b[2] - b[0]))


def _merge_vchains(words):
    """竖排的单字先拼成词块（'定'+'时'+'器' -> '定时器'）。

    不合并的后果：'通信接口' 会被拆成 4 行 "通/信/接/口"，
    每行一个孤立的字，检索和阅读都废掉。

    判据：两个字都是单个汉字、水平重叠 >= 60%、垂直间距 <= 8pt -> 属于同一列竖排。

    返回 [(x0, y0, x1, y1, text, is_chain)]。
    """
    singles = [w for w in words if CJK1.match(w[4])]
    others = [w for w in words if not CJK1.match(w[4])]
    singles.sort(key=lambda w: (round(w[0]), w[1]))

    used = [False] * len(singles)
    out = []
    for i, w in enumerate(singles):
        if used[i]:
            continue
        used[i] = True
        chain = [w]
        while len(chain) < MAX_VCHAIN:
            nxt = None
            for j, v in enumerate(singles):
                if used[j]:
                    continue
                if _overlap_h(chain[-1], v) >= 0.6 and 0 <= v[1] - chain[-1][3] <= 8:
                    nxt = j
                    break
            if nxt is None:
                break
            used[nxt] = True
            chain.append(singles[nxt])
        if len(chain) == 1:
            out.append((chain[0][0], chain[0][1], chain[0][2], chain[0][3], chain[0][4], False))
        else:
            out.append((min(c[0] for c in chain), min(c[1] for c in chain),
                        max(c[2] for c in chain), max(c[3] for c in chain),
                        "".join(c[4] for c in chain), True))
    out += [(w[0], w[1], w[2], w[3], w[4], False) for w in others]
    return out


def _find_row(y, row_y):
    for i in range(len(row_y) - 1):
        if row_y[i] - 1.0 <= y < row_y[i + 1]:
            return i
    return 0


def _find_col(span, col_x):
    """按"水平重叠最大"归列。

    为什么不是"完全落在列内"：实测 'ADC/TKey（通道数)' 的 x 是 91.2~178.8，
    横跨两列，用"完全包含"判据会被两个列都拒收 -> 整格文字凭空消失。
    """
    best, best_ov = 0, float("-inf")
    for j in range(len(col_x) - 1):
        ov = min(span[1], col_x[j + 1]) - max(span[0], col_x[j])
        if ov > best_ov:
            best, best_ov = j, ov
    return best


def _grid_of(page, table, words=None):
    boxes = [c for r in table.rows for c in r.cells if c]
    if not boxes:
        return None

    col_x = _merge_close([b[0] for b in boxes] + [b[2] for b in boxes], GC)
    row_y = _merge_close([b[1] for b in boxes] + [b[3] for b in boxes], GR)
    if len(col_x) < 2 or len(row_y) < 2:
        return None

    if words is None:
        words = page.get_text("words")
    x0t, y0t, x1t, y1t = table.bbox
    inside = [w for w in words
              if w[2] > x0t - 2 and w[0] < x1t + 2 and w[3] > y0t - 2 and w[1] < y1t + 2]
    tokens = _merge_vchains(inside)

    row_h = median([row_y[i + 1] - row_y[i] for i in range(len(row_y) - 1)])
    pad = row_h * 0.5

    nrow, ncol = len(row_y) - 1, len(col_x) - 1
    grid = [["" for _ in range(ncol)] for _ in range(nrow)]
    bbox_map = [[None] * ncol for _ in range(nrow)]
    span_map = [[None] * ncol for _ in range(nrow)]

    for (wx0, wy0, wx1, wy1, txt, is_chain) in tokens:
        txt = clean_cell(txt)
        if not txt:
            continue
        top, bot = (wy0, wy1) if wy0 <= wy1 else (wy1, wy0)
        # 竖排词块整体上移半个行高：竖排文字是垂直居中的，首字往往落在
        # 所属合并单元格的下一个行带上（实测 '定' 落在 '高级' 带、'通' 落在 'I2C' 带）。
        # 不外扩的话 '定时器' 会从 '高级' 行才开始、'通信接口' 会漏掉 'SPI' 行。
        if is_chain:
            top -= pad
            bot += pad
        i = _find_row(top, row_y)
        j = _find_col((wx0, wx1), col_x)

        grid[i][j] = (grid[i][j] + " " + txt).strip() if grid[i][j] else txt
        if bbox_map[i][j]:
            b = bbox_map[i][j]
            bbox_map[i][j] = (min(b[0], wx0), min(b[1], top), max(b[2], wx1), max(b[3], bot))
        else:
            bbox_map[i][j] = (wx0, top, wx1, bot)
        if span_map[i][j]:
            s = span_map[i][j]
            span_map[i][j] = (min(s[0], wx0), max(s[1], wx1))
        else:
            span_map[i][j] = (wx0, wx1)

    return grid, col_x, row_y, bbox_map, span_map


def _fill_down(grid, row_y, bbox_map):
    """纵向合并单元格继承：上方文字延伸到本行才继承。

    为什么不能"上一行非空就继承"：
        表1-1 的 '定时器' 覆盖 4 行，但下一行 'ADC/TKey（通道数)' 属于新分组。
        只看"上一行非空"会把 '定时器' 一直继承下去，污染后面几十行。
        加"文字 bbox 底边 >= 本行顶边"这一条，就正好在分组边界处断开。
    """
    nrow, ncol = len(grid), len(grid[0])
    for i in range(1, nrow):
        for j in range(ncol):
            if grid[i][j]:
                continue
            for i2 in range(i - 1, -1, -1):
                if not grid[i2][j]:
                    continue
                src = bbox_map[i2][j]
                if src is None:
                    break
                if src[3] >= row_y[i]:
                    grid[i][j] = grid[i2][j]
                    bbox_map[i][j] = src
                break
    return grid


# ---------- 二、表头 / 标签列的判定 ----------

def _header_count(grid):
    """表头有几行。

    判据：从第 0 行起，"数值单元格占比 >= 50%" 的那一行是数据行。
    为什么不用"含数字就是数据行"：表1-1 的表头里写着 'CH32F103'，含数字但不是数据。
    所以要求"整格就是一个数值"（48 / 32K / 2），'CH32F103' 不匹配。
    """
    n = 0
    for r in grid[:MAX_HEADER_ROWS]:
        vals = [c for c in r if c]
        if not vals:
            break
        num = sum(1 for c in vals if VALUE_RE.match(c))
        if num and num * 2 >= len(vals):
            break
        n += 1
    return n


def _label_col_count(grid, nhead):
    """标签列有几列。

    判据：某列在数据行里"大部分为空"、右邻列"大部分非空" -> 它是分组标签列。
    表1-1 的 '定时器' 列只在 4/20 行有值 -> 是分组标签，所以标签区 = 2 列。
    引脚表（PB10/PB11...）首列每行都有值 -> 标签区 = 1 列。
    """
    nrow, ncol = len(grid), len(grid[0])
    nd = nrow - nhead
    if nd <= 0:
        return 1
    nlabel = 1
    while nlabel < min(MAX_LABEL_COLS, ncol - 1):
        c = nlabel - 1
        f_here = sum(1 for i in range(nhead, nrow) if grid[i][c]) / nd
        f_next = sum(1 for i in range(nhead, nrow) if grid[i][c + 1]) / nd
        if f_here < 0.6 <= f_next:
            nlabel += 1
        else:
            break
    return nlabel


def _col_path(grid, nhead, j):
    """把多级表头的各层拼成一个列名：'CH32F103' + 'C6T6' -> 'CH32F103 C6T6'。"""
    parts = []
    for i in range(nhead):
        v = grid[i][j]
        if v and (not parts or parts[-1] != v):
            parts.append(v)
    return " ".join(parts)


# ---------- 三、对外入口 ----------

def rebuild_table_rows(page, table, words=None):
    """把 PyMuPDF 表格对象归一成规整二维行。

    返回 list[list[str]]：第 0 行是表头，第 1~ 行是数据；第 0 列是行标签。
    多列标签区（如 '定时器' + '通用'）会被拼成一格：'定时器 / 通用'。
    表头第一格是"行标签的名字"（如斜线表头 '产品型号 资源差异'）。
    """
    got = _grid_of(page, table, words)
    if not got:
        return []
    grid, col_x, row_y, bbox_map, span_map = got
    _fill_down(grid, row_y, bbox_map)

    ncol0 = len(grid[0])
    keep = [j for j in range(ncol0) if any(grid[i][j] for i in range(len(grid)))]
    if not keep:
        return []
    grid = [[r[j] for j in keep] for r in grid]
    span_map = [[span_map[i][j] for j in keep] for i in range(len(grid))]
    col_x = [col_x[j] for j in keep] + [col_x[keep[-1] + 1]]

    nhead = _header_count(grid)
    if nhead == 0 or nhead >= len(grid):
        nhead = 1
    nlabel = _label_col_count(grid, nhead)

    head_label = " ".join(dict.fromkeys(
        x for i in range(nhead) for x in grid[i][:nlabel] if x))
    ncol = len(grid[0])
    head_cells = [head_label] + [_col_path(grid, nhead, j) for j in range(nlabel, ncol)]

    out = [head_cells]
    for i in range(nhead, len(grid)):
        lab = LABEL_JOIN.join(x for x in grid[i][:nlabel] if x)
        vals = [grid[i][j] for j in range(nlabel, ncol)]
        if not lab and not any(vals):
            continue
        # 横向合并：整行只有一个数据格，而它的 x 跨过了列边界
        # -> 这是"跨列合并单元格"，值对每一列都成立（表1-1 的工作电压就是这样）
        idx = [k for k, v in enumerate(vals) if v]
        if len(idx) == 1 and len(vals) > 1:
            s = span_map[i][nlabel + idx[0]]
            if s and any(s[0] + 3 < col_x[k] < s[1] - 3
                         for k in range(nlabel + 1, len(col_x) - 1)):
                vals = [vals[idx[0]]] * len(vals)
        out.append([lab] + vals)
    return out


def rows_from_markdown(md: str):
    """把 markdown 表格读回二维行（chunk_loader 用，不需要 PDF 对象）。"""
    rows = []
    for ln in (md or "").splitlines():
        s = ln.strip()
        if not (s.startswith("|") and s.endswith("|")):
            continue
        if set(s.replace("|", "").strip()) <= set("-: "):
            continue
        rows.append([clean_cell(c) for c in s.strip("|").split("|")])
    return rows


# ---------- 四、序列化：每个数据点都带行头 + 列头 ----------

SMALL_TABLE_CHARS = 1500
GROUP_MAX_CHARS = 1500
CELL_SEP = " · "
KV_SEP = " = "


def table_cell_lines(rows):
    """每个非空数据格 -> 一行 "列头 · 行头 = 值"。

    这是你要的那件事："表格里的一个数据，连同它的表头信息一起抽出来"。
    例：CH32F103C6T6 那一格的 48 会变成
        CH32F103 C6T6 · 芯片引脚数 = 48
    —— 型号（列头）+ 属性（行头）+ 值，三样齐全，检索这一行就能答对。
    """
    if len(rows) < 2:
        return []
    head = rows[0]
    lines = []
    for r in rows[1:]:
        lab = (r[0] or "").strip()
        for k in range(1, len(r)):
            val = (r[k] or "").strip()
            if not val:
                continue
            col = (head[k] if k < len(head) else "").strip()
            key = (col + CELL_SEP + lab) if (col and lab) else (col or lab)
            lines.append(f"{key}{KV_SEP}{val}" if key else val)
    return lines


def table_to_text(rows, caption="", source="", page_label=""):
    """整表文本 = 父块内容（喂给 LLM 用，要全）。"""
    head = rows[0] if rows else []
    meta = [f"[表格] {caption}" if caption else "[表格]"]
    if source:
        meta.append(f"手册: {source}")
    if page_label:
        meta.append(f"位置: {page_label}")
    meta.append("列: " + " | ".join(h for h in head if h))
    return "\n".join(meta) + "\n\n" + "\n".join(table_cell_lines(rows))


def split_by_chars(lines, limit=GROUP_MAX_CHARS):
    """按字符数把行装桶（每桶 <= limit），用于把大表切成多个子块。"""
    out, cur, n = [], [], 0
    for ln in lines:
        if cur and n + len(ln) + 1 > limit:
            out.append(cur)
            cur, n = [], 0
        cur.append(ln)
        n += len(ln) + 1
    if cur:
        out.append(cur)
    return out
```

### 5.3 `is_heading`（第 369~392 行）整段替换

原实现的两个中文 bug：

1. `p.endswith((".", ",", ";", ":"))` —— 全是英文标点，中文的 `。，；：` 不在里面，  
   导致中文短句正文一路掉到最后一行被当成小节标题。
2. `len(p.split()) <= 12` —— 按空格分词，中文没空格，**整句话永远算 1 个词**，  
   这条判据对中文恒成立，等于只看"是不是 ≤60 字"。

```python
# 句子结束标点：中英文都要
SENT_END = ("。", "！", "？", "；", "：", "，", "、", "）", "】", "」", "》",
            ".", ",", ";", ":", ")", "]", "}")
# 表 / 图的题注（中英）
CAPTION_RE = re.compile(r"^(Table|Figure|表|图)\s*\d+([-.]\d+)*")


def is_heading(p: str) -> bool:
    """判断一个段落是不是"小节标题"。

    中文的两条额外规则：
      1. 结尾是中文句末标点 -> 是正文，不是标题
      2. 无编号的长短判据用【字符数】而不是【空格分词数】，中文上限 24 字
    """
    p = p.strip()
    if not p or "\n" in p:
        return False
    n = len(p)
    if n > 90:
        return False
    if SECTION_NUM_RE.match(p) and not PURE_NUMERIC_RE.fullmatch(p):
        return True
    if CAPTION_RE.match(p):
        return True
    if p.endswith(SENT_END):
        return False
    if has_cjk(p):
        return n <= 24                    # 中文：24 字以内才算标题
    return n <= 60 and len(p.split()) <= 12   # 英文：保持原判据
```

### 5.5 `make_table_chunks`（第 285~362 行）整段替换

新版本做三件事：

1. 把表格转成"每格一行"的 key-value（用户要的粒度）
2. 子块按字符数分桶（每桶 ≤ 1500），**父块 = 整张表**
3. 每个子块记 `parent_id`，供检索时放大块

```python
def make_table_chunks(blocks: list[dict], doc_name: str) -> tuple[list[dict], list[dict], list[str]]:
    """把合并后的表变成 chunk。

    返回 (子块列表, 父块列表, 备注)

    粒度设计（重要）：
        子块 = 每个数据点一行 "列头 · 行头 = 值"，按字符数分桶（每桶 <= 1500）
              -> 进向量库。问 "C6T6 的 SRAM 多大" 能直接命中那一行。
        父块 = 整张表的完整序列化
              -> 只在命中后喂给 LLM，保证模型看得到全表、能横向对比。
        这就是 Small-to-Big：检索用小（准），生成用大（全）。
    """
    chunks: list[dict] = []
    parents: list[dict] = []
    notes: list[str] = []

    for bi, b in enumerate(blocks, 1):
        title = b["title"] or "(无标题)"
        pages = b["pages"]
        page_label = (f"第 {pages[0]} 页" if len(pages) == 1
                      else f"第 {pages[0]}-{pages[-1]} 页（跨页续表，已合并）")

        rows = rows_from_markdown(b["md"])
        if len(rows) < 2:
            notes.append(f"- ⚠️ 《{title}》（{page_label}）重建后只剩表头/空行，已跳过")
            continue

        cell_lines = table_cell_lines(rows)
        if not cell_lines:
            notes.append(f"- ⚠️ 《{title}》（{page_label}）没有数据格，已跳过")
            continue

        full_text = table_to_text(rows, title, doc_name, page_label)
        pid = f"P_{doc_name}_{pages[0]}_{bi:04d}"

        parents.append({
            "id": pid,
            "type": "table",
            "source": doc_name,
            "title": title,
            "page_start": pages[0],
            "page_end": pages[-1],
            "text": full_text,
            "n_chars": len(full_text),
        })

        buckets = ([cell_lines] if len("\n".join(cell_lines)) <= SMALL_TABLE_CHARS
                   else split_by_chars(cell_lines))
        head_line = f"[{doc_name} {page_label} · 表格] {title}"

        for i, bucket in enumerate(buckets, 1):
            part = f"（第 {i}/{len(buckets)} 部分）" if len(buckets) > 1 else ""
            content = f"{head_line}{part}\n列: " + \
                      " | ".join(h for h in rows[0] if h) + "\n\n" + "\n".join(bucket)
            chunks.append({
                "content": content,          # 喂 LLM 的片段（= 子块本身）
                "index_text": content,       # 进向量的文字（表格不做 LLM 摘要！数值不能改写）
                "source": doc_name,
                "type": "table",
                "title": title,
                "page_start": pages[0],
                "page_end": pages[-1],
                "part": f"{i}/{len(buckets)}",
                "parent_id": pid,
                "n_chars": len(content),
            })

        if len(pages) > 1:
            notes.append(f"- 《{title}》横跨原 {pages} 页，已合并为完整一张")
        if len(buckets) > 1:
            notes.append(f"- 《{title}》序列化后 {len(cell_lines)} 行，"
                         f"切成 {len(buckets)} 个检索子块（父块保留全表）")

    return chunks, parents, notes
```

```python
def build_text_chunks(text_dir: Path, doc_name: str) -> tuple[list[dict], list[dict]]:
    """把正文页切成一堆小块，并同时产出"小节级父块"。

    两条关键规则（直接决定检索质量）：
      1. 遇到新的小节标题 -> 立刻收尾当前块
      2. 每块开头贴上下文前缀：[手册名 第 N 页 · 正文 · 这一节的标题路径]

    中英文用不同长度阈值：英文 500 字符 ≈ 80 词；中文 500 字 = 500 个汉字。
    """
    chunks: list[dict] = []
    parents: list[dict] = []
    files = sorted(text_dir.glob("page_*.md"))
    h1 = h2 = ""            # 跨页保留的章节上下文
    pid_seq = 0

    for f in files:
        meta, body = read_page_file(f)
        page = int(meta.get("page") or 0)
        paragraphs = [p.strip() for p in body.split("\n\n") if p.strip()]

        buf: list[str] = []
        cur_pid: str | None = None
        cur_section: list[str] = []

        def target_chars() -> int:
            txt = "\n".join(buf)
            return TEXT_TARGET_CHARS_CN if has_cjk(txt) else TEXT_TARGET_CHARS_EN

        def min_chars() -> int:
            txt = "\n".join(buf)
            return TEXT_MIN_CHARS_CN if has_cjk(txt) else TEXT_MIN_CHARS_EN

        def flush(cur_h1: str, cur_h2: str) -> None:
            nonlocal pid_seq, cur_pid, cur_section
            if not buf:
                return
            path = " > ".join(x for x in (cur_h1, cur_h2) if x)
            head = f"[{doc_name} 第 {page} 页 · 正文"
            if path:
                head += f" · {path}"
            head += "]"
            content = f"{head}\n\n" + "\n\n".join(buf)

            # 父块：同一个小节的子块共用一个 parent_id
            if cur_pid is None:
                pid_seq += 1
                cur_pid = f"P_{doc_name}_{page}_{pid_seq:04d}"
                cur_section = []
            cur_section.append(content)

            chunks.append({
                "content": content,
                "index_text": content,     # 本轮不做 LLM 摘要；留字段是为了下一轮直接加
                "source": doc_name,
                "type": "text",
                "title": path,
                "page_start": page,
                "page_end": page,
                "part": "1/1",
                "parent_id": cur_pid,
                "n_chars": len(content),
            })
            buf.clear()

        for p in paragraphs:
            if is_heading(p):
                if sum(len(x) + 2 for x in buf) >= min_chars():
                    flush(h1, h2)
                    # 换节 -> 开新父块
                    parents.append({
                        "id": cur_pid, "type": "text", "source": doc_name,
                        "title": " > ".join(x for x in (h1, h2) if x),
                        "page_start": page, "page_end": page,
                        "text": "\n\n".join(cur_section),
                        "n_chars": len("\n\n".join(cur_section)),
                    })
                    cur_pid = None
                h1, h2 = update_headings(p, h1, h2)
                buf.append(p)
                continue

            buf.append(p)
            if sum(len(x) + 2 for x in buf) >= target_chars():
                flush(h1, h2)

        flush(h1, h2)

    # 收尾：最后一个父块
    if cur_pid and cur_section:
        parents.append({
            "id": cur_pid, "type": "text", "source": doc_name,
            "title": "", "page_start": page, "page_end": page,
            "text": "\n\n".join(cur_section),
            "n_chars": len("\n\n".join(cur_section)),
        })
    return chunks, parents
```

## 7. 新增 `src/retrieval.py`：BM25 + RRF + 重排

```python
# -*- coding: utf-8 -*-
"""混合检索（BM25 + 向量）-> RRF 融合 -> 重排。

为什么需要 BM25：
    dense embedding 对"型号串和数字"不敏感 —— 你问 "CH32F103C6T6"
    和问 "CH32F103C8T6" 得到的向量几乎一样近。
    BM25 是精确词匹配，专门补这一刀。

为什么用 RRF 而不是"加权平均分数"：
    FAISS 给的是 L2 距离（越小越好），BM25 给的是词频分数（越大越好），
    量纲完全不同，没法直接相加。名次是唯一可以直接比较的东西。
    RRF：得分 = Σ 1/(k + rank)，两路都靠前的块自动浮上来。

为什么 RRF 之后还要重排：
    RRF 只用了"名次"这一维信息，粒度粗。重排模型会真正读一遍
    "问题 × 每块文字"，给出细粒度相关性分数。实测工业界组合
    （BEIR 基准 NDCG@10）：dense 52 -> +BM25/RRF 56 -> +rerank 62。
"""
from __future__ import annotations

import os
import re

import requests
from langchain_community.retrievers import BM25Retriever
from langchain_core.documents import Document

RRF_K = 60                 # 原论文默认值，不用调
FETCH_MULTIPLIER = 3       # 每条腿取 最终k 的几倍（RRF 需要足够的融合素材）
RERANK_MAX_DOC_CHARS = 1200

# 英文/数字/型号连成一个 token；中文按单字切。
# 为什么按单字切中文：中文没空格，按空格切等于整句算一个词，BM25 直接失效。
# 不引 jieba 是为了少一个依赖 —— 对"型号 + 数字"这类查询，按字切已经够用。
TOKEN_RE = re.compile(r"[a-zA-Z0-9_.\-+~/]+|[\u4e00-\u9fff]")


def tokenize(text: str) -> list[str]:
    return TOKEN_RE.findall((text or "").lower())


def doc_key(doc: Document) -> str:
    md = doc.metadata
    return f"{md.get('source', '?')}::{md.get('id') or doc.page_content[:60]}"


def rrf_fuse(rank_lists, k: int = RRF_K, weights=None, top_n=None):
    """Reciprocal Rank Fusion：把多路排名的名次倒数和加起来。"""
    scores: dict[str, float] = {}
    pool: dict[str, Document] = {}
    for i, results in enumerate(rank_lists):
        w = (weights or [1.0] * len(rank_lists))[i]
        for rank, doc in enumerate(results, start=1):
            key = doc_key(doc)
            scores[key] = scores.get(key, 0.0) + w / (k + rank)
            pool.setdefault(key, doc)
    ordered = sorted(scores.items(), key=lambda kv: kv[1], reverse=True)
    if top_n:
        ordered = ordered[:top_n]
    return [pool[key] for key, _ in ordered]


class HybridRetriever:
    """两路召回 + RRF。对外只有 invoke(query, source) -> list[Document]。"""

    def __init__(self, docs: list[Document], vectorstore,
                 k: int = 10, fetch_k: int | None = None,
                 weights: tuple[float, float] = (1.0, 1.0)):
        self.docs = docs
        self.vs = vectorstore
        self.k = k
        self.fetch_k = fetch_k or max(k * FETCH_MULTIPLIER, k + 20)
        self.weights = list(weights)
        self._bm25_all = BM25Retriever.from_documents(
            docs, preprocess_func=tokenize, k=self.fetch_k)
        self._by_source: dict[str, BM25Retriever] = {}

    def _bm25(self, source: str | None):
        """BM25Retriever 不支持 metadata 过滤，所以按手册各建一份并缓存。

        这一点和 FAISS 不一样：FAISS 的 similarity_search 有 filter 参数，
        BM25 没有，只能靠"预先只喂某一本手册的块"实现过滤。
        """
        if source is None:
            return self._bm25_all
        if source not in self._by_source:
            subset = [d for d in self.docs if d.metadata.get("source") == source]
            self._by_source[source] = BM25Retriever.from_documents(
                subset, preprocess_func=tokenize, k=self.fetch_k)
        return self._by_source[source]

    def invoke(self, query: str, source: str | None = None) -> list[Document]:
        flt = {"source": source} if source else None
        dense = self.vs.similarity_search(query, k=self.fetch_k, filter=flt)
        sparse = self._bm25(source).invoke(query)
        return rrf_fuse([sparse, dense], k=RRF_K,
                        weights=self.weights, top_n=self.k)


class ZhipuReranker:
    """智谱重排 API。失败时静默退化为"取前 N 条"，不阻断问答。"""

    URL = "https://open.bigmodel.cn/api/paas/v4/rerank"
    MODEL = "rerank"

    def __init__(self, api_key: str | None = None, timeout: int = 30):
        self.key = api_key or os.getenv("ZHIPUAI_API_KEY")
        self.timeout = timeout

    def rerank(self, query: str, docs: list[Document], top_n: int = 3) -> list[Document]:
        if not docs:
            return []
        if not self.key or len(docs) <= 1:
            return docs[:top_n]

        payload = {
            "model": self.MODEL,
            "query": query,
            "top_n": min(top_n, len(docs)),
            # 截断是为了避开接口的长度上限；1200 字足够判断相关性
            "documents": [d.page_content[:RERANK_MAX_DOC_CHARS] for d in docs],
        }
        try:
            r = requests.post(
                self.URL,
                headers={"Authorization": f"Bearer {self.key}",
                         "Content-Type": "application/json"},
                json=payload, timeout=self.timeout)
            r.raise_for_status()
            data = r.json()
            items = data.get("results") or data.get("data") or []
            out = []
            for it in items:
                idx = it.get("index")
                if isinstance(idx, int) and 0 <= idx < len(docs):
                    out.append(docs[idx])
            return out[:top_n] if out else docs[:top_n]
        except Exception as e:                       # noqa: BLE001
            print(f"[rerank] 调用失败，退化为按原顺序取前 {top_n} 条：{e}")
            return docs[:top_n]


def expand_to_parents(hits: list[Document], parents: dict[str, str]) -> list[str]:
    """Small-to-Big：检索用子块（准），喂给模型用父块（全）。

    去重是必须的 —— 同一小节的 5 个子块都命中时，
    不能把同一个父块往 prompt 里塞五遍。
    """
    out, seen = [], set()
    for d in hits:
        pid = d.metadata.get("parent_id")
        text = parents.get(pid) if pid else None
        if text:
            if pid in seen:
                continue
            seen.add(pid)
            out.append(text)
        else:
            key = doc_key(d)
            if key in seen:
                continue
            seen.add(key)
            out.append(d.page_content)
    return out
```
